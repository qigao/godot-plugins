cmake_minimum_required(VERSION 3.27)
project(GDMP LANGUAGES CXX)

set(gdmp_cpu "${CMAKE_SYSTEM_PROCESSOR}")
if(ANDROID)
  set(gdmp_platform android)
  set(gdmp_cpu "${CMAKE_ANDROID_ARCH_ABI}")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(gdmp_platform macos)
  if(CMAKE_OSX_ARCHITECTURES)
    set(gdmp_cpu "${CMAKE_OSX_ARCHITECTURES}")
  endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(gdmp_platform linux)
elseif(WIN32)
  set(gdmp_platform windows)
  if(NOT MINGW OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "The Windows MediaPipe SDK requires LLVM-MinGW")
  endif()
else()
  message(FATAL_ERROR "No MediaPipe SDK for ${CMAKE_SYSTEM_NAME}")
endif()

string(TOLOWER "${gdmp_cpu}" gdmp_cpu)
if(gdmp_cpu MATCHES "^(aarch64|arm64|arm64-v8a)$")
  set(gdmp_arch arm64)
  set(sdk_arch arm64)
elseif(gdmp_cpu MATCHES "^(x86_64|amd64)$")
  set(gdmp_arch x86_64)
  set(sdk_arch x64)
else()
  message(FATAL_ERROR "GDMP requires a single x64 or ARM64 target, got '${gdmp_cpu}'")
endif()

find_package(MediaPipe 1.2.2 EXACT CONFIG REQUIRED)
if(NOT MediaPipe_SDK_PLATFORM STREQUAL "${gdmp_platform}-${sdk_arch}")
  message(FATAL_ERROR "GDMP requires the MediaPipe SDK for ${gdmp_platform}-${sdk_arch}")
endif()
if(WIN32)
  if(MediaPipe_GPU_ENABLED OR NOT MediaPipe_CPU_DELEGATE STREQUAL "xnnpack")
    message(FATAL_ERROR "GDMP on Windows requires the CPU SDK with XNNPACK")
  endif()
elseif(NOT MediaPipe_GPU_ENABLED)
  message(FATAL_ERROR "GDMP requires the GPU-enabled MediaPipe SDK for ${gdmp_platform}")
endif()
if(ANDROID)
  set(gdmp_android_api "${CMAKE_SYSTEM_VERSION}")
  if(DEFINED ANDROID_PLATFORM_LEVEL)
    set(gdmp_android_api "${ANDROID_PLATFORM_LEVEL}")
  endif()
  if(gdmp_android_api VERSION_LESS MediaPipe_ANDROID_API)
    message(FATAL_ERROR "The MediaPipe SDK requires Android API ${MediaPipe_ANDROID_API} or later")
  endif()
endif()
if(gdmp_platform STREQUAL "macos" AND CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS MediaPipe_MACOS_MINIMUM)
  message(FATAL_ERROR "Set CMAKE_OSX_DEPLOYMENT_TARGET to ${MediaPipe_MACOS_MINIMUM} or later for the MediaPipe SDK")
endif()
set(GODOT_CPP_DIR "" CACHE PATH "Godot C++ bindings at e83fd0904c13356ed1d4c3d09f8bb9132bdc6b77")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/godot-cpp-profile.json")
set(GODOTCPP_TARGET template_release)
set(GODOTCPP_SYSTEM_HEADERS ON)
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp EXCLUDE_FROM_ALL)
set_property(SOURCE "${GODOT_CPP_DIR}/src/godot.cpp" TARGET_DIRECTORY godot-cpp
  APPEND PROPERTY COMPILE_OPTIONS -include cstdlib)

file(GLOB_RECURSE gdmp_sources CONFIGURE_DEPENDS
  GDMP/framework/*.cc
  GDMP/gpu/*.cc
  GDMP/proto/*.cc
  GDMP/tasks/*.cc
  GDMP/util/*.cc)
# MediaPipe 1.2.2 removed the Face Stylizer Tasks API.
list(FILTER gdmp_sources EXCLUDE REGEX "/face_stylizer[.]cc$")
add_library(gdmp SHARED
  GDMP/gdlibrary.cc
  GDMP/io/camera_helper.cc
  GDMP/io/camera_helper_stub.cc
  ${gdmp_sources})
target_compile_features(gdmp PRIVATE cxx_std_20)
target_compile_definitions(gdmp PRIVATE GDMP_MEDIAPIPE_SDK_1_2_2=1)
target_include_directories(gdmp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gdmp PRIVATE MediaPipe::Tasks godot::cpp)
if(gdmp_platform STREQUAL "macos")
  target_link_options(gdmp PRIVATE "LINKER:-exported_symbol,_mediapipe_library_init")
elseif(WIN32)
  target_link_options(gdmp PRIVATE -static-libstdc++ -static-libgcc)
else()
  set(export_map "${CMAKE_CURRENT_SOURCE_DIR}/GDMP/exports.lds")
  target_link_options(gdmp PRIVATE "LINKER:--version-script=${export_map}" "LINKER:-z,defs")
  set_property(TARGET gdmp APPEND PROPERTY LINK_DEPENDS "${export_map}")
endif()
if(ANDROID)
  target_sources(gdmp PRIVATE GDMP/android/asset_util.cc GDMP/android/jni.cc)
  target_link_options(gdmp PRIVATE "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
endif()

set(addon "${CMAKE_CURRENT_BINARY_DIR}/addons/GDMP")
set_target_properties(gdmp PROPERTIES
  OUTPUT_NAME GDMP.${gdmp_platform}
  LIBRARY_OUTPUT_DIRECTORY "${addon}/libs/${gdmp_arch}")
if(WIN32)
  set_target_properties(gdmp PROPERTIES PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${addon}/libs/${gdmp_arch}")
endif()
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../../addons/GDMP/" DESTINATION "${addon}")
get_filename_component(mediapipe_sdk "${MediaPipe_DIR}/../../.." ABSOLUTE)
file(COPY "${mediapipe_sdk}/licenses" DESTINATION "${addon}")
configure_file(LICENSE "${addon}/LICENSE" COPYONLY)
configure_file("${GODOT_CPP_DIR}/LICENSE.md" "${addon}/LICENSE.godot-cpp" COPYONLY)
