cmake_minimum_required(VERSION 3.27)
project(SaltsQRCode LANGUAGES C CXX)

set(qrcode_cpu "${CMAKE_SYSTEM_PROCESSOR}")
if(ANDROID)
  set(qrcode_platform android)
  set(qrcode_cpu "${CMAKE_ANDROID_ARCH_ABI}")
elseif(APPLE)
  set(qrcode_platform macos)
  if(CMAKE_OSX_ARCHITECTURES)
    set(qrcode_cpu "${CMAKE_OSX_ARCHITECTURES}")
  endif()
elseif(WIN32)
  set(qrcode_platform windows)
  if(NOT MINGW OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Salts QR Code on Windows requires LLVM-MinGW")
  endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(qrcode_platform linux)
else()
  message(FATAL_ERROR "Salts QR Code targets Linux, Windows, macOS and Android")
endif()

string(TOLOWER "${qrcode_cpu}" qrcode_cpu)
if(qrcode_cpu MATCHES "^(arm64|aarch64|arm64-v8a)$")
  set(qrcode_arch arm64)
elseif(qrcode_cpu MATCHES "^(x86_64|amd64)$")
  set(qrcode_arch x86_64)
else()
  message(FATAL_ERROR "Salts QR Code requires a single x64 or ARM64 target, got '${qrcode_cpu}'")
endif()

set(GODOT_CPP_DIR "" CACHE PATH "godot-cpp source at 507ed9d840c01a3c5b2a39af8bb4000bfac30bf5")
set(ZXING_CPP_DIR "" CACHE PATH "ZXing-C++ 3.1.1 release source")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/godot-cpp-profile.json")
set(GODOTCPP_API_VERSION 4.7)
set(GODOTCPP_TARGET template_release)
set(GODOTCPP_SYSTEM_HEADERS ON)
set(GODOTCPP_DISABLE_EXCEPTIONS OFF)
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp EXCLUDE_FROM_ALL)

set(BUILD_SHARED_LIBS OFF)
set(ZXING_READERS ON)
set(ZXING_WRITERS NEW)
set(ZXING_ENABLE_QRCODE ON)
foreach(format IN ITEMS 1D AZTEC DATAMATRIX MAXICODE PDF417)
  set("ZXING_ENABLE_${format}" OFF)
endforeach()
set(ZXING_USE_BUNDLED_ZINT ON)
add_subdirectory("${ZXING_CPP_DIR}/core" zxing-cpp EXCLUDE_FROM_ALL)

add_library(salts_qrcode SHARED salts_qrcode.cpp)
target_compile_features(salts_qrcode PRIVATE cxx_std_20)
target_link_libraries(salts_qrcode PRIVATE godot::cpp ZXing::ZXing)
if(APPLE)
  target_link_options(salts_qrcode PRIVATE "LINKER:-exported_symbol,_salts_qrcode_library_init")
elseif(WIN32)
  target_link_options(salts_qrcode PRIVATE -static-libstdc++ -static-libgcc)
  set_target_properties(salts_qrcode PROPERTIES PREFIX "")
else()
  target_link_options(salts_qrcode PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:-z,defs")
endif()
if(ANDROID)
  target_link_options(salts_qrcode PRIVATE "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
endif()

set(addon_source "${CMAKE_CURRENT_SOURCE_DIR}/../../addons/salts_qrcode")
set(addon "${CMAKE_CURRENT_BINARY_DIR}/addons/salts_qrcode")
set(bin "${addon}/bin/${qrcode_platform}/${qrcode_arch}")
set_target_properties(salts_qrcode PROPERTIES
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${bin}/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${bin}/$<0:>"
  ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/import-libraries/$<0:>")
configure_file("${addon_source}/salts_qrcode.gdextension" "${addon}/salts_qrcode.gdextension" COPYONLY)
configure_file("${addon_source}/README.md" "${addon}/README.md" COPYONLY)
configure_file("${addon_source}/THIRD_PARTY_NOTICES.txt" "${addon}/THIRD_PARTY_NOTICES.txt" COPYONLY)
configure_file("${ZXING_CPP_DIR}/LICENSE" "${addon}/LICENSE.ZXing-C++" COPYONLY)
configure_file("${GODOT_CPP_DIR}/LICENSE.md" "${addon}/LICENSE.godot-cpp" COPYONLY)

option(BUILD_TESTING "Run the Salts QR Code Godot smoke test" OFF)
if(BUILD_TESTING)
  enable_testing()
  set(GODOT_EXECUTABLE "" CACHE FILEPATH "Godot 4.7.2 executable for headless tests")
  if(NOT EXISTS "${GODOT_EXECUTABLE}")
    message(FATAL_ERROR "Set GODOT_EXECUTABLE to the Godot 4.7.2 executable")
  endif()
  set(tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/salts_qrcode")
  configure_file("${tests}/project.godot" project.godot COPYONLY)
  configure_file("${tests}/qrcode_smoke.gd" qrcode_smoke.gd COPYONLY)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.godot/extension_list.cfg"
    "res://addons/salts_qrcode/salts_qrcode.gdextension\n")
  add_test(NAME salts_qrcode.godot
    COMMAND "${GODOT_EXECUTABLE}" --headless --path "${CMAKE_CURRENT_BINARY_DIR}" --script res://qrcode_smoke.gd)
  set_tests_properties(salts_qrcode.godot PROPERTIES TIMEOUT 30)
endif()
